FROM node:20-slim AS builder

WORKDIR /usr/src/app
COPY package*.json tsconfig.json ./
RUN npm ci

COPY client/ ./client/
RUN npx tsc -p tsconfig.json

FROM node:20-slim

WORKDIR /usr/src/app
COPY package*.json ./
RUN npm ci --only=production

COPY --from=builder /usr/src/app/dist ./dist

USER node
ENV NODE_ENV=production

EXPOSE 8080
CMD ["node", "dist/daemon.js"]
